**humanoid.py Router 範例：**

```python
# backend/routers/humanoid.py

from fastapi import APIRouter, HTTPException
from typing import List
import logging

from services.humanoid_service import HumanoidService
from models.requests import GeneratePersonaRequest
from models.responses import PersonaResponse

router = APIRouter()
logger = logging.getLogger(__name__)

# 依賴注入（將在後續優化）
humanoid_service = HumanoidService()

@router.post("/generate", response_model=PersonaResponse)
async def generate_persona(request: GeneratePersonaRequest):
    """生成新的角色人格"""
    try:
        persona = await humanoid_service.generate(request)
        return PersonaResponse(success=True, data=persona)
    except Exception as e:
        logger.error(f"生成角色失敗: {e}")
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/list", response_model=List[PersonaResponse])
async def list_personas():
    """列出所有角色"""
    personas = await humanoid_service.get_all()
    return personas

@router.get("/{persona_id}", response_model=PersonaResponse)
async def get_persona(persona_id: str):
    """獲取特定角色"""
    persona = await humanoid_service.get_by_id(persona_id)
    if not persona:
        raise HTTPException(status_code=404, detail="角色不存在")
    return PersonaResponse(success=True, data=persona)

@router.delete("/{persona_id}")
async def delete_persona(persona_id: str):
    """刪除角色"""
    success = await humanoid_service.delete(persona_id)
    if not success:
        raise HTTPException(status_code=404, detail="角色不存在")
    return {"success": True, "message": "角色已刪除"}
```

## 五、階段 3：現代化改造（可選）

### 5.1 為什麼需要現代化？

目前的 `frontend.html`（1346行）存在問題：
- ❌ 單一 HTML 檔案過於龐大

// src/components/PersonaList.jsx
import { useState, useEffect } from 'react'
import { humanoidAPI } from '../api/humanoid'
import PersonaCard from './PersonaCard'

function PersonaList() {
  const [personas, setPersonas] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  useEffect(() => {
    loadPersonas()
  }, [])

  const loadPersonas = async () => {
    try {
      setLoading(true)
      const data = await humanoidAPI.getAll()
      setPersonas(data)
    } catch (err) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  const handleDelete = async (id) => {
    try {
      await humanoidAPI.delete(id)
      setPersonas(personas.filter(p => p.id !== id))
    } catch (err) {
      alert('刪除失敗: ' + err.message)
    }
  }

  if (loading) return <div>載入中...</div>
  if (error) return <div>錯誤: {error}</div>

  return (
    <div className="persona-list">
      {personas.map(persona => (
        <PersonaCard
          key={persona.id}
          persona={persona}
          onDelete={() => handleDelete(persona.id)}
        />
      ))}
    </div>
  )
}

export default PersonaList

```javascript
// src/stores/personaStore.js
import { create } from 'zustand'
import { humanoidAPI } from '../api/humanoid'

export const usePersonaStore = create((set, get) => ({
  personas: [],
  selectedPersona: null,
  loading: false,
  error: null,

  // 載入所有角色
  loadPersonas: async () => {
    set({ loading: true, error: null })
    try {
      const data = await humanoidAPI.getAll()
      set({ personas: data, loading: false })
    } catch (error) {
      set({ error: error.message, loading: false })
    }
  },

  // 生成新角色
  generatePersona: async () => {
    set({ loading: true })
    try {
      const newPersona = await humanoidAPI.generate()
      set(state => ({
        personas: [...state.personas, newPersona],
        loading: false
      }))
      return newPersona
    } catch (error) {
      set({ error: error.message, loading: false })
      throw error
    }
  },

  // 刪除角色
  deletePersona: async (id) => {
    try {
      await humanoidAPI.delete(id)
      set(state => ({
        personas: state.personas.filter(p => p.id !== id)
      }))
    } catch (error) {
      set({ error: error.message })
      throw error
    }
  },

  // 選擇角色
  selectPersona: (persona) => set({ selectedPersona: persona }),
}))
```

## 八、常見問題與解答

### Q1：為什麼要前後端分離？
**A：** 
- ✅ 前端可以獨立開發和部署
- ✅ 支援使用現代前端框架
- ✅ 提高開發效率（熱重載）
- ✅ 便於團隊分工
- ✅ 方便橫向擴展

### Q2：分離後性能會變差嗎？
**A：** 不會，甚至會更好：
- 前端可以使用 CDN
- 後端可以專注於 API 優化
- 可以分別擴展前後端服務

### Q3：需要重寫所有前端程式碼嗎？
**A：** 不需要：
- 階段 1-2 可以保留原有 HTML/JS
- 階段 3 是可選的現代化改造
- 可以逐步遷移，不必一次全改

### Q4：Unity WebGL 怎麼處理？
**A：** 兩種選擇：
1. 保留在後端（簡單）
2. 遷移到前端專案（推薦）

### Q5：舊的 API 會受影響嗎？
**A：** 不會：
- 只是移除 HTML 路由
- API 端點保持不變
- Unity 客戶端無需修改
